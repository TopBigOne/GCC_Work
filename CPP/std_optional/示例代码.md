### **示例 1：基础用法**
```cpp
#include <iostream>
#include <optional>
#include <string>

// 可能返回有效值或空的可选函数
std::optional<int> safe_divide(int a, int b) {
    if (b == 0) return std::nullopt;
    return a / b;
}

int main() {
    // 基本声明方式
    std::optional<int> optInt;
    std::optional<std::string> optStr = "Hello";
    
    // 检查是否有值
    if (optStr.has_value()) {
        std::cout << *optStr << std::endl;  // 输出 Hello
    }

    // 安全除法示例
    auto result1 = safe_divide(10, 2);
    auto result2 = safe_divide(5, 0);

    std::cout << "Result1: " << result1.value_or(-1) << std::endl;  // 5
    std::cout << "Result2: " << result2.value_or(-1) << std::endl;  // -1

    try {
        std::cout << result2.value();  // 抛出 bad_optional_access
    } catch (const std::bad_optional_access& e) {
        std::cerr << "Error: " << e.what() << std::endl;
    }
}
```

### **示例 2：查找元素**
```cpp
#include <vector>
#include <algorithm>
#include <optional>

template <typename T>
std::optional<size_t> find_index(const std::vector<T>& vec, const T& target) {
    auto it = std::find(vec.begin(), vec.end(), target);
    if (it != vec.end()) {
        return std::distance(vec.begin(), it);
    }
    return std::nullopt;
}

int main() {
    std::vector<int> numbers{1, 3, 5, 7, 9};
    
    if (auto index = find_index(numbers, 5)) {
        std::cout << "Found at index: " << *index << std::endl;  // 输出 2
    } else {
        std::cout << "Not found" << std::endl;
    }

    if (auto index = find_index(numbers, 4)) {
        // 不会执行
    } else {
        std::cout << "4 Not found" << std::endl;
    }
}
```

### **示例 3：延迟初始化**
```cpp
#include <optional>
#include <iostream>

class ExpensiveResource {
    // 假设这是需要大量初始化时间的资源
public:
    ExpensiveResource() { 
        std::cout << "Resource initialized\n"; 
    }
    ~ExpensiveResource() { 
        std::cout << "Resource released\n"; 
    }
    void use() { 
        std::cout << "Using resource\n"; 
    }
};

class ResourceHolder {
    std::optional<ExpensiveResource> resource_;
    
public:
    void lazy_init() {
        if (!resource_) {
            resource_.emplace();  // 延迟初始化
        }
    }
    
    void use_resource() {
        if (resource_) {
            resource_->use();
        } else {
            std::cout << "Resource not initialized\n";
        }
    }
};

int main() {
    ResourceHolder holder;
    holder.use_resource();  // 输出 Resource not initialized
    
    holder.lazy_init();     // 输出 Resource initialized
    holder.use_resource();  // 输出 Using resource
    
    // 离开作用域时输出 Resource released
}
```

### **示例 4：配置解析**
```cpp
#include <optional>
#include <string>
#include <sstream>

struct Config {
    std::optional<int> brightness;
    std::optional<double> timeout;
    std::optional<std::string> theme;
};

std::optional<Config> parse_config(const std::string& input) {
    std::istringstream iss(input);
    Config cfg;
    std::string key;
    
    while (iss >> key) {
        if (key == "brightness:") {
            int value;
            if (iss >> value) cfg.brightness = value;
        }
        else if (key == "timeout:") {
            double value;
            if (iss >> value) cfg.timeout = value;
        }
        else if (key == "theme:") {
            std::string value;
            if (iss >> value) cfg.theme = value;
        }
    }
    
    // 至少需要设置一个值才认为配置有效
    if (cfg.brightness || cfg.timeout || cfg.theme) {
        return cfg;
    }
    return std::nullopt;
}

int main() {
    auto cfg = parse_config("brightness: 80 theme: dark timeout: 15.5");
    
    if (cfg) {
        std::cout << "Parsed config:\n";
        if (cfg->brightness) 
            std::cout << "Brightness: " << *cfg->brightness << "\n";
        if (cfg->timeout) 
            std::cout << "Timeout: " << *cfg->timeout << "\n";
        if (cfg->theme) 
            std::cout << "Theme: " << *cfg->theme << "\n";
    }
}
```

### **示例 5：安全访问嵌套数据**
```cpp
#include <optional>
#include <iostream>
#include <vector>

struct Person {
    std::string name;
    std::optional<std::string> email;
};

struct Department {
    std::string name;
    std::optional<Person> manager;
};

struct Company {
    std::vector<Department> departments;
};

std::optional<std::string> find_ceo_email(const Company& company) {
    for (const auto& dept : company.departments) {
        if (dept.name == "Executive") {
            if (dept.manager) {
                if (dept.manager->email) {
                    return dept.manager->email;
                }
            }
        }
    }
    return std::nullopt;
}

int main() {
    Company company{
        {
            {"IT", Person{"Alice", "alice@company.com"}},
            {"HR", Person{"Bob", std::nullopt}},
            {"Executive", Person{"Carol", "ceo@company.com"}}
        }
    };

    if (auto email = find_ceo_email(company)) {
        std::cout << "CEO email: " << *email << "\n";
    } else {
        std::cout << "CEO email not found\n";
    }
}
```

### **示例 6：与 STL 算法结合**
```cpp
#include <vector>
#include <optional>
#include <algorithm>
#include <iostream>

std::optional<int> first_even(const std::vector<int>& nums) {
    auto it = std::find_if(nums.begin(), nums.end(),
        [](int n) { return n % 2 == 0; });
    
    if (it != nums.end()) return *it;
    return std::nullopt;
}

int main() {
    std::vector<int> data{1, 3, 5, 7, 8, 9};
    
    if (auto num = first_even(data)) {
        std::cout << "First even number: " << *num << "\n";  // 8
    } else {
        std::cout << "No even numbers found\n";
    }

    std::vector<int> odd_only{1, 3, 5, 7, 9};
    std::cout << "First even in odd set: " 
              << first_even(odd_only).value_or(-1) << "\n";  // -1
}
```

### **示例 7：C++23 新特性（如果编译器支持）**
```cpp
#if __cplusplus >= 202002L
#include <optional>
#include <iostream>

std::optional<int> double_if_positive(int x) {
    if (x > 0) return x * 2;
    return std::nullopt;
}

int main() {
    std::optional<int> val = 5;
    
    // 使用 transform（映射）
    auto squared = val.transform([](int x) { return x * x; });
    std::cout << squared.value() << "\n";  // 25
    
    // 使用 and_then（链式调用）
    auto result = val.and_then(double_if_positive)
                 .and_then([](int x) { return x + 10; });
    std::cout << result.value() << "\n";  // 20 (5*2 + 10)
    
    // 处理空值
    std::optional<int> empty;
    std::cout << empty.and_then(double_if_positive)
                     .transform([](int x) { return x / 2; })
                     .value_or(-1) << "\n";  // -1
}
#endif
```

### **关键点总结**
1. **创建方式**：
    - `std::optional<T> var;`（空值）
    - `var = value;` 或 `var.emplace(args...);`
    - `return std::nullopt;` 返回空值

2. **安全访问**：
    - `if (opt)` 或 `has_value()` 检查存在性
    - `value()` 获取值（带检查）
    - `value_or(default)` 安全获取
    - `operator*` 和 `operator->`（需确保值存在）

3. **最佳实践**：
    - 优先使用 `value_or()` 而不是直接解引用
    - 在函数返回时优先于返回特殊值
    - 与结构化绑定结合使用
    - 注意大对象的复制成本

这些示例展示了 `std::optional` 在不同场景下的典型用法，建议实际运行并修改这些示例来加深理解。
